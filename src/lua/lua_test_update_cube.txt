function Start()
  print "Starting CUBE from LUA!"
  Rotate(0.0, 0.0, 0.0)
  AddComponent(0)
end

local rotation = 0
function Update()
  print "Updating CUBE from LUA!"
  Translate(math.sin(os.clock()) * 5.0, math.cos(os.clock()) * 5.0, 0.0)
  Scale(math.sin(os.clock()) * 2.0, math.cos(os.clock()) * 2.0, math.sin(os.clock()) * 2.0)
end

function Start()
  print "Starting SPHERE from LUA!"
  SetGeometry("Cube")
  SetDrawMode("Triangles")
end

local rotation_ = 0
local speed_ = 10
function Rotation()
  rotation_ = rotation_ + 0.01
  Rotate(0.0, rotation_, 0.0)
end

function Update()
  --print "Updating SPHERE from LUA!"
  Rotation()
end
